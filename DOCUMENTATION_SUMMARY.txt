================================================================================
                         WORKFLOW QUICK GUIDE
================================================================================

TO ANALYZE A SENTRY ISSUE:
  1. cd /Users/ishan.shah/projects/ai-alert
  2. ./ai-alert-ssl analyze <issue-id>
  3. View result on screen

TO VIEW PREVIOUS ANALYSES:
  1. poetry run ai-alert list
  2. poetry run ai-alert show <alert-id>

TO EXPORT ALL ANALYSES:
  sqlite3 ~/.ai-alert/data.db ".mode json" \
    "SELECT * FROM analyses;" > /tmp/analyses.json

TO SET UP SLACK DAEMON:
  poetry run ai-alert daemon start --foreground

TO BATCH ANALYZE ISSUES:
  for id in 7691078133 7691078134 7691078135; do
    ./ai-alert-ssl analyze $id
  done

================================================================================
                           COMMON COMMANDS
================================================================================

ANALYZE:
  ./ai-alert-ssl analyze 7691078133              # Basic analysis
  ./ai-alert-ssl analyze 7691078133 --force      # Force re-analyze
  ./ai-alert-ssl analyze 7691078133 --dry-run    # Show prompt only

LIST & SHOW:
  poetry run ai-alert list                       # All alerts
  poetry run ai-alert list --status pending      # Pending only
  poetry run ai-alert list --status analyzed     # Completed only
  poetry run ai-alert show <alert-id>            # Show full analysis

STATUS & CONFIG:
  poetry run ai-alert status                     # Status overview
  poetry run ai-alert config-check               # Validate config

DAEMON & OPERATIONS:
  poetry run ai-alert daemon start --foreground  # Start listening to Slack
  poetry run ai-alert gc --days 30               # Delete old analyses

DATABASE:
  sqlite3 ~/.ai-alert/data.db "SELECT COUNT(*) FROM analyses;"
  sqlite3 ~/.ai-alert/data.db ".mode json" "SELECT * FROM analyses;"

LOGS:
  tail -f ~/.ai-alert/logs/ai-alert-*.log
